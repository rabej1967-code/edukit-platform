import { createClient } from "jsr:@supabase/supabase-js@2"; const RESEND_API_KEY = Deno.env.get("RESEND_API_KEY")!; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SUPABASE_SERVICE_KEY = Deno.env.get("SERVICE_ROLE_KEY")!; function generatePassword(length = 10): string { const chars = "ABCDEFGHJKMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789"; let pw = ""; for (let i = 0; i < length; i++) { pw += chars.charAt(Math.floor(Math.random() * chars.length)); } return pw; } function getPakej(amount: number): string { if (amount === 9500) return "Pakej Guru"; if (amount === 16500) return "Pakej Guru + Pentadbir"; if (amount === 300) return "KidSpark Sewaan Bulanan"; return "Pakej Pentadbir"; } async function sendWelcomeEmail(email: string, password: string, pakej: string) { try { const res = await fetch("https://api.resend.com/emails", { method: "POST", headers: { "Authorization": `Bearer ${RESEND_API_KEY}`, "Content-Type": "application/json", }, body: JSON.stringify({ from: "EduKit ", to: [email], subject: "🎉 Selamat Datang ke EduKit — Maklumat Log Masuk Anda", html: `

© 2025 EduKit — Platform Digital untuk Warga Pendidik Malaysia

`, }), }); const json = await res.json(); console.log("Email result:", JSON.stringify(json)); } catch (e) { console.error("Email error:", e); } } Deno.serve(async (req) => { try { const body = await req.text(); let event; try { event = JSON.parse(body); } catch { return new Response(JSON.stringify({ received: true }), { headers: { "Content-Type": "application/json" }, }); } console.log("Event:", event.type); // Hanya proses checkout.session.completed if (event.type !== "checkout.session.completed") { return new Response(JSON.stringify({ received: true, skipped: true }), { headers: { "Content-Type": "application/json" }, }); } const session = event.data.object; const email = session.customer_details?.email || session.customer_email || session.receipt_email; const amount = session.amount_total || 0; const pakej = getPakej(amount); console.log("Email:", email, "Amount:", amount, "Pakej:", pakej); if (!email) { return new Response(JSON.stringify({ received: true }), { headers: { "Content-Type": "application/json" }, }); } const sb = createClient(SUPABASE_URL, SUPABASE_SERVICE_KEY); const password = generatePassword(); let userId: string | undefined; // Cuba create user baru const { data: created, error: createErr } = await sb.auth.admin.createUser({ email, password, email_confirm: true, }); if (created?.user) { userId = created.user.id; console.log("New user:", userId); } else if (createErr) { console.log("User exists:", createErr.message); // Cari dari profiles table const { data: profile } = await sb .from("profiles") .select("id") .eq("email", email) .single(); if (profile?.id) { userId = profile.id; await sb.auth.admin.updateUserById(userId, { password }); console.log("Password reset for:", userId); } } // Update profile if (userId) { const pkgId = pakej.includes("Pentadbir") && pakej.includes("Guru") ? "gabungan" : pakej.includes("Guru") ? "guru" : "pentadbir"; await sb.from("profiles").upsert({ id: userId, email, package_id: pkgId, must_change_password: true, default_pw: password, }); console.log("Profile updated:", pkgId); } // Hantar email await sendWelcomeEmail(email, password, pakej); return new Response(JSON.stringify({ received: true, success: true }), { headers: { "Content-Type": "application/json" }, }); } catch (e) { console.error("Error:", e); // Sentiasa return 200 supaya Stripe tidak retry return new Response(JSON.stringify({ received: true }), { headers: { "Content-Type": "application/json" }, }); } });
